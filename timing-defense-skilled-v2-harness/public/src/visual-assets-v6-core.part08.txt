function scan() {
  if (!screen) return;
  if (!screen.classList.contains('view-game')) {
    cleanupSlideState();
    return;
  }
  if (slideState && isSolvedView()) {
    cleanupSlideState();
    return;
  }
  for (const board of screen.querySelectorAll('.board-wrap:not(.linkline-board-ghost)')) decorateBoard(board);
}

const observer = new MutationObserver(scan);
if (screen) observer.observe(screen, { childList: true, subtree: true });
queueMicrotask(scan);
document.addEventListener(DRAG_INTENT_EVENT, receiveDragIntent, { capture: true });
