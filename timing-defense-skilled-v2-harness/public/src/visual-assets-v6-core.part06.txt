    cleanupSlideState();
    return;
  }
  const stageKey = stageLabel?.textContent?.trim() || '';
  if (stageKey !== slideState.stageKey) {
    cleanupSlideState();
    return;
  }

  const board = currentBoard();
  if (board) reconcileFromBoard(board);
  // Transient board replacement never tears down the persistent drag layer.

  if (!slideState.finishing) {
    renderActiveFrame();
    ensureFrame();
    return;
  }

  const target = settleTargetPoints();
  if (!target?.length) {
    cleanupSlideState();
    return;
  }
  if (!slideState.settleFrom?.length) {
    slideState.settleFrom = slideState.renderPoints.map((point) => ({ ...point }));
    slideState.settleStartedAt = now;
  }
  const duration = reducedMotion ? 0 : SETTLE_MS;
  const t = duration === 0 ? 1 : Math.min(1, Math.max(0, (now - slideState.settleStartedAt) / duration));
  slideState.renderPoints = target.map((point, index) => lerpPoint(slideState.settleFrom[index] ?? point, point, t));
  updateOverlayGeometry(slideState.renderPoints);
  if (t < 1) ensureFrame();
  else cleanupSlideState();
}

function beginSharedIntent(intent) {
  cleanupSlideState();
  if (!screen?.classList.contains('view-game') || isSolvedView()) return;
  const board = currentBoard();
  const info = boardInfo(board);
  const source = sourcePathSamples(board);
  if (!info || !source) return;
  decorateBoard(board);
  const ordered = orderChainCellsBySource(board, info.n, source.points, intent.endpointCell);
  if (!ordered?.length) return;
  const geometry = authoritativeGeometry(ordered, info.n);
  if (!geometry?.metrics?.total) return;

  const overlay = createPersistentOverlay(info.n, ordered.length);
  slideState = {
    pointerId: intent.pointerId,
    stageKey: stageLabel?.textContent?.trim() || '',
    n: info.n,
    authoritativeHeadCell: intent.endpointCell,
    chainCells: ordered,
    chainLength: geometry.metrics.total,
    intent,
    renderPoints: geometry.points,
    overlay,
    boardRect: info.rect,
    lastBoard: board,
    raf: 0,
    finishing: false,
    settleFrom: null,
    settleStartedAt: 0,
  };
  document.documentElement.classList.add('linkline-v61-drag');
  positionPersistentOverlay(info.rect);
  updateOverlayGeometry(slideState.renderPoints);
  ensureFrame();
}

function receiveDragIntent(event) {
  const detail = event?.detail;
  if (detail?.version !== 1) return;
  const intent = copyIntent(detail);
  if (!intent) return;

  if (intent.phase === 'start') {
    beginSharedIntent(intent);
    return;
  }
  if (!slideState || intent.pointerId !== slideState.pointerId) return;

  if (!applyCommittedHead(intent.endpointCell)) {
    // A malformed/ambiguous commit never mutates presentation state.
    return;
  }
  slideState.intent = intent;

  if (intent.phase === 'finish') {
    slideState.finishing = true;
    slideState.settleFrom = slideState.renderPoints.map((point) => ({ ...point }));
    slideState.settleStartedAt = performance.now();
  }
  ensureFrame();
}
