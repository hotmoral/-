1
        den=cum[j+1]-cum[j]; a=0.0 if den==0 else (t-cum[j])/den
        out[i]=points[j]*(1-a)+points[j+1]*a
    return out


def smooth_polyline(points: np.ndarray):
    p=points.copy(); k=np.array([1,4,6,4,1],float)/16.0
    for _ in range(PATH_SMOOTH_PASSES):
        q=p.copy()
        for i in range(2,len(p)-2): q[i]=(p[i-2:i+3]*k[:,None]).sum(axis=0)
        q[0]=p[0]; q[-1]=p[-1]; p=q
    return p


def select_internal_points(points: np.ndarray, count: int, fallback: Iterable[float]):
    p=smooth_polyline(resample_polyline(points)); n=len(p); cand=[]
    for i in range(2,n-2):
        frac=i/(n-1)
        if not (CURVATURE_ENDPOINT_GUARD_FRACTION <= frac <= 1-CURVATURE_ENDPOINT_GUARD_FRACTION): continue
        a=p[i]-p[i-2]; b=p[i+2]-p[i]; na=np.linalg.norm(a); nb=np.linalg.norm(b)
        if na<=0 or nb<=0: continue
        angle=math.acos(float(np.clip(np.dot(a,b)/(na*nb),-1,1)))
        if angle >= CURVATURE_MIN_RAD: cand.append((angle,i,frac))
    selected=[]
    for score,i,frac in sorted(cand,key=lambda x:(-x[0],x[1])):
        if all(abs(frac-s[2])>=CURVATURE_MIN_SEPARATION_FRACTION for s in selected):
            selected.append((score,i,frac))
            if len(selected)==count: break
    if len(selected)<count:
        qs=tuple(fallback)
        if len(qs)!=count: raise RuntimeError("fallback quantile count mismatch")
        return [(float(q), p[int(round(float(q)*(n-1)))].copy(), "FALLBACK_ARC_QUANTILE") for q in qs]
    return [(frac,p[i].copy(),"CURVATURE_EXTREMUM") for _score,i,frac in sorted(selected,key=lambda x:x[2])]


def point_at_fraction(points: np.ndarray, fraction: float) -> np.ndarray:
    p=resample_polyline(points); return p[int(round(fraction*(len(p)-1)))].copy()


def derive_semantic_graph(vertices, graph, transform, pitch):
    anchors=derive_surface_anchors(vertices)
    snapped={k:nearest_graph_node(v,graph,transform) for k,v in anchors.items()}
    protected=set(snapped.values())
    g=prune_graph(graph,protected,pitch,body_metrics(vertices)["height"])
    head=snapped["HEAD"]
    fl,fr=snapped["FORE_L_PAW"],snapped["FORE_R_PAW"]
    hl,hr=snapped["HIND_L_PAW"],snapped["HIND_R_PAW"]
    tail=snapped["TAIL_TIP"]
    hfl,_=shortest_path(g,head,fl,pitch); hfr,_=shortest_path(g,head,fr,pitch)
    hhl,_=shortest_path(g,head,hl,pitch); hhr,_=shortest_path(g,head,hr,pitch)
    fore=pair_merge_node(g,hfl,hfr,fl,fr,pitch)
    pelvis=pair_merge_node(g,hhl,hhr,hl,hr,pitch)
    pelvis_head,_=shortest_path(g,pelvis,head,pitch)
    tail_head,_=shortest_path(g,tail,head,pitch)
    tail_root=first_merge_from_tail(g,tail_head,pelvis_head,tail,pitch)
    paths={
        "PELVIS_FOREQUARTER": shortest_path(g,pelvis,fore,pitch)[0],
        "FOREQUARTER_HEAD": shortest_path(g,fore,head,pitch)[0],
        "FORE_L": shortest_path(g,fore,fl,pitch)[0], "FORE_R": shortest_path(g,fore,fr,pitch)[0],
        "HIND_L": shortest_path(g,pelvis,hl,pitch)[0], "HIND_R": shortest_path(g,pelvis,hr,pitch)[0],
        "TAIL": shortest_path(g,tail_root,tail,pitch)[0],
    }
    return g, snapped, {"PELVIS":pelvis,"FOREQUARTER":fore,"TAIL_ROOT":tail_root,"HEAD":head}, paths
